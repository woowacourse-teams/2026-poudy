package com.poudy.product.repository;

import com.poudy.product.domain.ProductQuery;
import com.poudy.product.domain.ProductSort;
import com.poudy.product.repository.ProductQueryRepository;
import com.poudy.product.repository.ProductRepository;
import com.poudy.product.repository.ProductLoader;
import com.poudy.product.repository.ProductAttributeRepository;
import com.poudy.brand.repository.BrandRepository;
import java.time.Clock;
import org.springframework.jdbc.core.RowMapper;
import org.springframework.jdbc.core.namedparam.NamedParameterJdbcTemplate;
import org.springframework.jdbc.core.namedparam.SqlParameterSource;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Arrays;
import java.util.List;
import java.util.function.LongSupplier;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.test.context.jdbc.Sql;
import org.springframework.transaction.annotation.Transactional;

@SpringBootTest
@Transactional
@Sql({"/db/search-test-data.sql", "/db/product-suggestion-and-test-data.sql"})
class Search818BenchmarkTest {
    @Autowired ProductQueryRepository queries;
    @Autowired ProductRepository products;
    @Autowired JdbcTemplate jdbc;

    @Autowired ProductLoader loader;
    @Autowired ProductAttributeRepository attributes;
    @Autowired BrandRepository brands;
    @Autowired Clock clock;

    @Test
    void measure() throws Exception {
        String phase = System.getenv("POUDY_818_PHASE");
        Path output = Path.of("/tmp/poudy-search-818");
        StringBuilder measurements = new StringBuilder("route\tkeyword\tphase\tround\tsample\tdurationMs\tresultCount\n");
        ProductQueryRepository before = new ProductQueryRepository(new NamedParameterJdbcTemplate(jdbc) {
            @Override
            public <T> List<T> query(String sql, SqlParameterSource parameters, RowMapper<T> mapper) {
                return super.query(sql.replace("search_products(:keyword, :offset, :size, 2)", "search_products(:keyword, :offset, :size, 3)"), parameters, mapper);
            }
        }, loader, brands, clock, attributes);
        ProductQueryRepository after = new ProductQueryRepository(new NamedParameterJdbcTemplate(jdbc), loader, brands, clock, attributes);
        StringBuilder metadata = new StringBuilder();
        for (String setting : List.of("server_version", "server_encoding", "shared_buffers", "work_mem", "max_parallel_workers_per_gather", "jit", "statement_timeout")) {
            metadata.append(setting).append('\t').append(jdbc.queryForObject("show " + setting, String.class)).append('\n');
        }
        metadata.append("lc_collate\t").append(jdbc.queryForObject("select datcollate from pg_database where datname = current_database()", String.class)).append('\n');
        metadata.append("products\t").append(jdbc.queryForObject("select count(*) from product", Long.class)).append('\n');
        metadata.append("java\t").append(System.getProperty("java.version")).append('\n');
        metadata.append("os\t").append(System.getProperty("os.name")).append(' ').append(System.getProperty("os.arch")).append('\n');
        Files.writeString(output.resolve("benchmark-" + phase + "-metadata.tsv"), metadata);
        StringBuilder plans = new StringBuilder();
        for (String keyword : List.of("로션", "에스트라 로션", "로션 닥터지 앰플", "에스트라 로", "ㅂ", "피디알엔", "검증스내일", "라운드랩검증독도", "검증토너")) {
            boolean filtered = keyword.equals("검증토너");
            int page = filtered ? 2 : 1;
            List<Long> brands = filtered ? List.of(90000L) : null;
            ProductQuery query = new ProductQuery(keyword, null, brands, null, null, null, null, null, null, null, null, false, List.of());
            for (String route : List.of("SUGGEST", "CATALOG", "COUNT", "HAS_RESULTS")) {
                LongSupplier oldOperation = operation(before, route, keyword, query, page);
                LongSupplier newOperation = operation(after, route, keyword, query, page);
                for (int i = 0; i < 50; i++) { oldOperation.getAsLong(); newOperation.getAsLong(); }
                for (int round = 1; round <= 5; round++) {
                    for (int i = 0; i < 100; i++) {
                        for (boolean oldFirst : (i % 2 == 0 ? List.of(true, false) : List.of(false, true))) {
                            LongSupplier selected = oldFirst ? oldOperation : newOperation;
                            long started = System.nanoTime();
                            long count = selected.getAsLong();
                            double elapsed = (System.nanoTime() - started) / 1000000.0;
                            measurements.append(route).append('\t').append(keyword).append('\t').append(oldFirst ? "before" : "after").append('\t')
                                .append(round).append('\t').append(i).append('\t').append(elapsed).append('\t').append(count).append('\n');
                        }
                    }
                }
            }
            int maxTier = 2;
            plans.append("keyword=").append(keyword).append("; maxTier=").append(maxTier).append('\n');
            plans.append(jdbc.queryForObject("explain (analyze, buffers, format json) select * from search_products(?, ?, 20, ?)", String.class, keyword, (page - 1) * 20, maxTier)).append('\n');
        }
        Files.writeString(output.resolve("benchmark-" + phase + ".tsv"), measurements);
        Files.writeString(output.resolve("benchmark-" + phase + "-plans.txt"), plans);
    }

    private LongSupplier operation(ProductQueryRepository selected, String route, String keyword, ProductQuery query, int page) {
        return switch (route) {
            case "SUGGEST" -> () -> selected.suggest(keyword, page, 20).totalElements();
            case "CATALOG" -> () -> selected.find(query, ProductSort.DEFAULT, page, 20).totalElements();
            case "COUNT" -> () -> selected.count(query);
            default -> () -> products.hasSearchResults(keyword) ? 1 : 0;
        };
    }
}
